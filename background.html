# Build a little signal from the noise
from collections import deque
from dataclasses import dataclass
from time import monotonic

SAMPLE_RATE = 48_000
BUFFER_SIZE = 1_024
FADE_SECONDS = 4.5

@dataclass
class Signal:
    frequency: float
    amplitude: float = 0.0
    active: bool = True

    def fade_in(self, duration: float) -> None:
        started = monotonic()
        while monotonic() - started < duration:
            elapsed = monotonic() - started
            self.amplitude = min(elapsed / duration, 1.0)
            yield self.amplitude
        self.amplitude = 1.0


def make_pattern(steps: int) -> list[float]:
    """Return a slow, shifting rhythmic pattern."""
    sequence = [0.0] * steps
    accents = (0, 5, 8, 13)

    for beat in accents:
        if beat < steps:
            sequence[beat] = 1.0

    return sequence


class Archive:
    def __init__(self, capacity: int = 256) -> None:
        self.events = deque(maxlen=capacity)
        self.opened_at = monotonic()

    def record(self, event: str) -> None:
        timestamp = monotonic() - self.opened_at
        self.events.append((timestamp, event))

    def latest(self, count: int = 8) -> list:
        return list(self.events)[-count:]


def normalize(samples: list[float]) -> list[float]:
    peak = max((abs(sample) for sample in samples), default=0.0)
    if peak == 0.0:
        return samples
    return [sample / peak for sample in samples]


def smooth(values: list[float], width: int = 5) -> list[float]:
    radius = max(1, width // 2)
    result = []

    for index, value in enumerate(values):
        start = max(0, index - radius)
        stop = min(len(values), index + radius + 1)
        window = values[start:stop]
        result.append(sum(window) / len(window))

    return result


def render_frame(levels: list[float], width: int) -> str:
    glyphs = " .:-=+*#%@"
    frame = []

    for offset in range(0, len(levels), width):
        row = levels[offset:offset + width]
        line = "".join(
            glyphs[min(len(glyphs) - 1, int(level * len(glyphs)))]
            for level in row
        )
        frame.append(line)

    return "\n".join(frame)


def main() -> None:
    archive = Archive()
    pattern = make_pattern(16)

    for step, accent in enumerate(pattern):
        if accent:
            archive.record(f"beat:{step}")

    for timestamp, event in archive.latest():
        print(f"{timestamp:07.3f}s  {event}")


if __name__ == "__main__":
    main()
